Model Type.BELONGING bm {
    initial Disengaged
    state Prepared
    state Passive
    state Active

    transition Disengaged -> Prepared: prepare for SoS
    transition Prepared -> Disengaged: disengage from SoS
    transition Prepared -> Passive: leave SoS
    transition Passive -> Prepared: join SoS
    transition Passive -> Active: join constellation
    transition Active -> Passive: leave constellation
}
