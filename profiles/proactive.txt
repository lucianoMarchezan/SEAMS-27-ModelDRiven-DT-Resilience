Reliability proactive extends adaptive {
    Belonging model: bm-proactive

    policy constraint "Failed systems cannot participate"  {
            when hm = Failed and bm-proactive in {Active, Passive}
            then require bm-proactive = Prepared
    }

    policy constraint "Failed systems cannot be admitted" {
            when hm = Failed
            then deny transition bm-proactive::Prepared -> bm-proactive::Passive
    }

    policy adaptation "Stabilize entry/exit" {
        step 1 {
            when bm-proactive = PendingEntry
            then maintain for stabilization_time
        }
        step 2 {
            when bm-proactive = PendingExit
            then maintain for stabilization_time
        }
    }
}
