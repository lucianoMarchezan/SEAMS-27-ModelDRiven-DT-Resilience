Model Type.HEALTH hm {
    initial Ideal
    states Defective, Faulty, Erroneous, Malfunctioning, Degraded, Failed

    transition component-deviation Ideal -> Defective
    transition defect-activated Defective -> Faulty
    transition fault-exercised Faulty -> Erroneous
    transition error-propagated Erroneous -> Malfunctioning
    transition impact-observed Malfunctioning -> Degraded
    transition threshold-violated Degraded -> Failed
    transition recovery {Defective, Faulty, Erroneous, Malfunctioning, Degraded, Failed} -> Ideal
}
