Model Type.BELONGING bm-proactive extends bm {
    refine Active {
        initial PendingEntry
        states PendingEntry, Participating, 
        PendingExit
    }
    
    transition stabilization-complete PendingEntry -> Participating
    transition leave-participation Participating -> PendingExit
    transition stop-active Active -> bm::Prepared
    override transition bm::leave-constellation PendingExit -> bm::Passive
}
